cmake_minimum_required(VERSION 3.24)
if(POLICY CMP0141)
  cmake_policy(SET CMP0141 NEW)
  set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
endif()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION.txt" OPENNEX_RELEASE LIMIT_COUNT 1)
project(OpenNex VERSION ${OPENNEX_RELEASE} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(JLU_BUILD_UI "Build Qt Quick desktop application" ON)
if(NOT WIN32)
  message(FATAL_ERROR "The Windows transport and credential store require Windows")
endif()
add_library(jlu_core src/core/protocol.cpp src/core/session.cpp src/platform/windows_network.cpp)
target_include_directories(jlu_core PUBLIC src)
target_compile_definitions(jlu_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(jlu_core PRIVATE /W4 /WX /utf-8)
target_link_libraries(jlu_core PUBLIC bcrypt ws2_32 iphlpapi advapi32)
include(CTest)
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/core_tests.cpp")
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE jlu_core)
  target_compile_options(core_tests PRIVATE /W4 /WX /utf-8)
  add_test(NAME offline_core COMMAND core_tests)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/traffic_tests.cpp")
  add_executable(traffic_tests tests/traffic_tests.cpp)
  target_include_directories(traffic_tests PRIVATE src)
  target_compile_options(traffic_tests PRIVATE /W4 /WX /utf-8)
  add_test(NAME offline_traffic COMMAND traffic_tests)
endif()
if(JLU_BUILD_UI)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Widgets Network)
  qt_standard_project_setup(REQUIRES 6.8)
  enable_language(RC)
  configure_file(packaging/windows.rc.in "${CMAKE_CURRENT_BINARY_DIR}/windows.rc" @ONLY)
  qt_add_executable(OpenNex WIN32 src/app/main.cpp src/app/controller.cpp src/app/controller.h src/app/speedtest.cpp src/app/speedtest.h src/app/updater.cpp src/app/updater.h src/app/update_manifest.cpp "${CMAKE_CURRENT_BINARY_DIR}/windows.rc")
  target_sources(OpenNex PRIVATE src/app/proxy_query.cpp)
  target_link_libraries(OpenNex PRIVATE jlu_core Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Network crypt32 powrprof shell32)
  target_compile_definitions(OpenNex PRIVATE OPENNEX_VERSION="${PROJECT_VERSION}")
  target_compile_options(OpenNex PRIVATE /W4 /utf-8)
  qt_add_resources(OpenNex "branding" PREFIX "/branding" BASE "assets/branding"
    FILES assets/branding/app.png assets/branding/mark.svg assets/branding/tray.svg)
  qt_add_resources(OpenNex "licenses" PREFIX "/licenses" FILES LICENSE THIRD_PARTY.md)
  qt_add_resources(OpenNex "updates" PREFIX "/updates" FILES packaging/update-public-key.hex tools/apply-update.ps1)
  qt_add_qml_module(OpenNex URI OpenNex VERSION 1.0
    QML_FILES ui/Main.qml ui/ActionButton.qml ui/Panel.qml ui/Field.qml ui/AppText.qml ui/AppIcon.qml ui/PreferenceRow.qml ui/InfoRow.qml ui/SelectField.qml ui/TrafficChart.qml ui/SpeedGauge.qml ui/SpeedChart.qml ui/SpeedPanel.qml)
  if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/controller_tests.cpp")
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(speed_tests tests/speed_tests.cpp src/app/speedtest.cpp src/app/speedtest.h)
    target_sources(speed_tests PRIVATE src/app/proxy_query.cpp)
    target_link_libraries(speed_tests PRIVATE Qt6::Test Qt6::Network Qt6::Widgets ws2_32)
    target_compile_definitions(speed_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(speed_tests PRIVATE /W4 /utf-8)
    add_test(NAME offline_speed COMMAND speed_tests)
    set_tests_properties(offline_speed PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
    qt_add_executable(speed_probe tests/speed_probe.cpp src/app/speedtest.cpp src/app/speedtest.h)
    target_sources(speed_probe PRIVATE src/app/proxy_query.cpp)
    target_link_libraries(speed_probe PRIVATE Qt6::Network Qt6::Widgets ws2_32)
    target_compile_definitions(speed_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(speed_probe PRIVATE /W4 /utf-8)
    qt_add_executable(update_tests tests/update_tests.cpp src/app/update_manifest.cpp)
    target_link_libraries(update_tests PRIVATE Qt6::Test bcrypt)
    target_compile_definitions(update_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_include_directories(update_tests PRIVATE src/app)
    target_compile_options(update_tests PRIVATE /W4 /utf-8)
    add_test(NAME offline_update COMMAND update_tests)
    set_tests_properties(offline_update PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
    qt_add_executable(controller_tests tests/controller_tests.cpp src/app/controller.cpp src/app/controller.h)
    qt_add_resources(controller_tests "licenses" PREFIX "/licenses" FILES LICENSE THIRD_PARTY.md)
    qt_add_executable(settings_fixture tests/settings_fixture.cpp)
    target_link_libraries(settings_fixture PRIVATE Qt6::Core)
    target_compile_options(settings_fixture PRIVATE /W4 /utf-8)
    target_link_libraries(controller_tests PRIVATE jlu_core Qt6::Test Qt6::Widgets Qt6::Network crypt32)
    target_compile_options(controller_tests PRIVATE /W4 /utf-8)
    add_test(NAME offline_controller COMMAND controller_tests)
    set_tests_properties(offline_controller PROPERTIES TIMEOUT 65
      ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QML_DISABLE_DISK_CACHE=1"
      ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
  endif()
  install(TARGETS OpenNex RUNTIME DESTINATION .)
  install(FILES LICENSE THIRD_PARTY.md README.md DESTINATION .)
endif()
